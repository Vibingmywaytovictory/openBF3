function(openbf3_module target)
    target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /fp:precise)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
endfunction()

function(openbf3_registry target dir header registry space prefix index)
    file(GLOB sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}/*.cpp")
    list(SORT sources)
    set(declarations "")
    set(calls "")
    foreach(source IN LISTS sources)
        get_filename_component(stem "${source}" NAME_WE)
        string(APPEND declarations "void ${prefix}${stem}(${registry} &r_registry);\n")
        string(APPEND calls "\t${prefix}${stem}(r_registry);\n")
    endforeach()
    set(parameter "${registry} &")
    if(NOT calls STREQUAL "")
        set(parameter "${registry} &r_registry")
    endif()
    set(text "\#include \"${header}\"\n\nnamespace ${space} {\n\n${declarations}\nvoid ${index}(${parameter}) {\n${calls}}\n\n}\n")
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/${index}.cpp")
    file(CONFIGURE OUTPUT "${generated}" CONTENT "${text}" @ONLY)
    target_sources(${target} PRIVATE ${sources} "${generated}")
endfunction()
